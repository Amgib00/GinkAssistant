# --- building the app ---
# Which Briefcase depends on the Mac (checked in each release's source):
#  * macOS 10.15 Catalina or older (Darwin < 20): 0.3.15 -- the newest whose apps open
#    there (Python package b6 + launcher built for 10.15) AND that builds into build/.
#    (0.3.12 and older build into ./macOS/; 0.3.16+ make apps that need macOS 11.)
#  * macOS 11+ / Windows / Linux: 0.4.4 on Python 3.10 (0.4.5+ needs Python 3.11), else newest.
briefcase==0.3.15; sys_platform == "darwin" and platform_release < "20"
briefcase==0.4.4; python_version < "3.11" and (sys_platform != "darwin" or platform_release >= "20")
briefcase>=0.4.4; python_version >= "3.11" and (sys_platform != "darwin" or platform_release >= "20")
